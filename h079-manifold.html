inductive Plane where
 | xy | xz | xw | yz | yw | zw
def turn (p : K) (a : Plane) : K :=
 let (x,y,z,w) := p
 match a with
 | .xy => (-y,x,z,w)
 | .xz => (-z,y,x,w)
 | .xw => (-w,y,z,x)
 | .yz => (x,-z,y,w)
 | .yw => (x,-w,z,y)
 | .zw => (x,y,-w,z)
def normSq (p : K) : ℝ :=
